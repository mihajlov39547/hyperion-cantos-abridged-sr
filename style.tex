    \vspace{6mm}
    {\color{HyperionRule}\rule{0.44\textwidth}{0.4pt}\par}
    \vspace{18mm}
    \IfFileExists{\CoverImagePath}{%
      \includegraphics[width=0.44\textwidth,height=46mm,keepaspectratio]{\CoverImagePath}\par
      \vspace{16mm}
    }{%
      {\fontsize{18}{22}\selectfont\color{HyperionMuted}\textasteriskcentered\quad
      \textasteriskcentered\quad\textasteriskcentered\par}
      \vspace{20mm}
    }
    \vfill
    {\small\scshape \BookCreditLabel\par}
    \vspace{2mm}
    {\large \BookCreditName\par}
    \vspace{14mm}
    {\large \BookYear\par}
    \vspace*{12mm}
  \end{titlepage}
  \pagecolor{white}
}
