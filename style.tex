\newcommand{\HyperionFullPageImage}[1]{%
  \begin{tikzpicture}[remember picture,overlay]
    \clip (current page.south west) rectangle (current page.north east);
    \node[inner sep=0pt,anchor=center] at (current page.center) {%
      \includegraphics[width=\paperwidth]{#1}%
    };
  \end{tikzpicture}%
}

\newcommand{\makehyperioncover}{%
  \hypersetup{pageanchor=false}%
  \pagenumbering{Alph}%
  \begin{titlepage}
    \thispagestyle{empty}
    \IfFileExists{\CoverImagePath}{%
      \HyperionFullPageImage{\CoverImagePath}%
    }{%
      \pagecolor{HyperionInk}%
    }%
    \begin{tikzpicture}[remember picture,overlay]
      \node[
        anchor=north,
        align=center,
        text=white,
        text width=0.82\paperwidth
      ] at ([yshift=-31mm]current page.north) {%
        {\fontsize{41}{45}\selectfont\scshape \BookTitle\par}
        \vspace{4mm}
        {\Large\scshape \BookSubtitle\par}
      };
      \node[
        anchor=south,
        align=center,
        text=white,
        text opacity=0.86,
        text width=0.78\paperwidth
      ] at ([yshift=18mm]current page.south) {%
        {\small\scshape \BookCreditLabel\par}
        \vspace{2mm}
        {\large \BookCreditName\par}
        \vspace{8mm}
        {\large \BookYear\par}
      };
    \end{tikzpicture}%
    \null
  \end{titlepage}
  \pagecolor{white}
  \hypersetup{pageanchor=true}%
}
\let\makehyperiontitle\makehyperioncover

\newcommand{\makehyperionback}{%
  \clearpage
  \hypersetup{pageanchor=false}%
  \thispagestyle{empty}
  \IfFileExists{\BackImagePath}{%
    \HyperionFullPageImage{\BackImagePath}%
  }{%
    \pagecolor{HyperionInk}%
  }%
  \begin{tikzpicture}[remember picture,overlay]
    \node[
      anchor=south,
      align=center,
      text=white,
      text opacity=0.82,
      text width=0.78\paperwidth
    ] at ([yshift=17mm]current page.south) {%
      {\Large\scshape \BookTitle\par}
      \vspace{3mm}
      {\large \BookYear\par}
    };
  \end{tikzpicture}%
  \null
  \clearpage
  \pagecolor{white}
  \hypersetup{pageanchor=true}%
}

    }{}%

  \fi
